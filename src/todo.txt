impl Ease<f64> for f64 {
    fn lerp(self, target: Self, s: f64) -> Self {
        self + (target - self) * s
    }

    fn damp(self, target: Self, lambda: f64, dt: f32) -> Self {
        self.lerp(target, 1.0 - f64::exp(-lambda * dt as f64))
    }

    fn step(self, edge: Self) -> Self {
        if self < edge { 0.0 } else { 1.0 }
    }
}

impl Ease<f32> for Vec2 {
    fn lerp(self, target: Self, s: f32) -> Self {
        Vec2::lerp(self, target, s)
    }

    fn damp(self, target: Self, lambda: f32, dt: f32) -> Self {
        Vec2::lerp(self, target, 1.0 - f32::exp(-lambda * dt))
    }

    fn step(self, edge: Self) -> Self {
        Vec2 {
            x: self.x.step(edge.x),
            y: self.y.step(edge.y),
        }
    }
}
