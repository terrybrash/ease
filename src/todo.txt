pub trait Ease<L> {
    /// Framerate independant "lerp" from `a` to `b`.
    /// It's not really a lerp but more of an exponential easing where `lambda` is the
    /// strength of the dampening. Can be greater than 1.0.
    /// https://www.rorydriscoll.com/2016/03/07/frame-rate-independent-damping-using-lerp/
    fn damp(self, target: Self, lambda: L, dt: f32) -> Self;
    fn lerp(self, target: Self, lambda: L) -> Self;
    fn step(self, x: Self) -> Self;
}

impl Ease<f64> for f64 {
    fn lerp(self, target: Self, s: f64) -> Self {
        self + (target - self) * s
    }

    fn damp(self, target: Self, lambda: f64, dt: f32) -> Self {
        self.lerp(target, 1.0 - f64::exp(-lambda * dt as f64))
    }

    fn step(self, edge: Self) -> Self {
        if self < edge { 0.0 } else { 1.0 }
    }
}

impl Ease<f32> for f32 {
    fn lerp(self, target: Self, s: f32) -> Self {
        self + (target - self) * s
    }

    fn damp(self, target: Self, lambda: f32, dt: f32) -> Self {
        self.lerp(target, 1.0 - f32::exp(-lambda * dt))
    }

    fn step(self, edge: Self) -> Self {
        if self < edge { 0.0 } else { 1.0 }
    }
}

impl Ease<f32> for Vec2 {
    fn lerp(self, target: Self, s: f32) -> Self {
        Vec2::lerp(self, target, s)
    }

    fn damp(self, target: Self, lambda: f32, dt: f32) -> Self {
        Vec2::lerp(self, target, 1.0 - f32::exp(-lambda * dt))
    }

    fn step(self, edge: Self) -> Self {
        Vec2 {
            x: self.x.step(edge.x),
            y: self.y.step(edge.y),
        }
    }
}

pub fn damp_radians(a: Number, b: Number, lambda: f32, dt: f32) -> Number {
    lerp_radians(a, b, 1.0 - f32::exp(-lambda * dt))
}

pub fn lerp_radians(a: Number, b: Number, x: Number) -> Number {
    let cs = (1.0 - x) * f32::cos(a) + x * f32::cos(b);
    let sn = (1.0 - x) * f32::sin(a) + x * f32::sin(b);
    f32::atan2(sn, cs)
}

/// Returns a value from `0..1` along a parabolic curve.
pub fn parabolic(x: Number) -> Number {
    1.0 - f32::powf(x * 2.0 - 1.0, 2.0)
}

/// Returns a value from `0..1` along a hyperbolic curve.
pub fn hyperbolic(rating: Number) -> Number {
    1.0 - 1.0 / (1.0 + rating)
}

/// `edge0` shouldn't be greater than or equal to `edge1`
pub fn smoothstep(edge0: Number, edge1: Number, x: Number) -> Number {
    let t = Number::clamp((x - edge0) / (edge1 - edge0), 0.0, 1.0);
    t * t * (3.0 - 2.0 * t)
}

pub trait Wrap {
    /// v is input value, m is modulus. wrap(a, b) is a replacement for a % b.
    // This won't work for negative or zero modulo values.
    // Try this instead (if necessary):
    // https://www.imaginary-institute.com/resources/TechNote12/TechNote12.html
    // https://github.com/rust-lang/rust/issues/87970
    // https://news.ycombinator.com/item?id=34540353
    fn wrap(self, m: Self) -> Self;
    fn wrap_every(duration: Self, )
}

impl Wrap for f32 {
    fn wrap(self, m: Self) -> Self {
        if m <= 0.0 { 0.0 } else { ((self % m) + m) % m }
    }

    fn wrap_every(duration: f32, now: f32) -> Self {
        f32::wrap(now, duration) / duration
    }
}

impl Wrap for f64 {
    fn wrap(self, m: Self) -> Self {
        if m <= 0.0 { 0.0 } else { ((self % m) + m) % m }
    }
}

impl Wrap for i32 {
    fn wrap(self, m: Self) -> Self {
        if m <= 0 { 0 } else { ((self % m) + m) % m }
    }
}
